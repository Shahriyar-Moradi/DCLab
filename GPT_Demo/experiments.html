Reproduction script · E7 generator v2 · reproduces CV, max |Δ| 0.0

# DCLab reproduction for experiment E7 (branch of E2) — no DCLab imports
# inputs: dataset 7f3a…c21, split_assignment a3f9…d10 (fold map), spec v1
# decisions: churned_at excluded (leak, d-302), region_code categorical (d-298, AI L2)
# user note: churn = "no payment for 60 days"
NUM = [...]; CAT = [..., "region_code"]; DATE = ["signup_date"]
prep = ColumnTransformer([
  ("num", Pipeline([("impute", SimpleImputer(strategy="median")), ("scale", StandardScaler())]), NUM),
  ("cat", Pipeline([("impute", SimpleImputer(strategy="most_frequent")),
                    ("onehot", OneHotEncoder(handle_unknown="ignore", drop=None))]), CAT),
  ("date", DateParts(["month", "weekday"]), DATE)])
model = Pipeline([("prep", prep),
  ("clf", LGBMClassifier(class_weight="balanced", num_leaves=31, learning_rate=0.05, n_estimators=420, random_state=42))])
# folds from the stored assignment map; holdout rows are never used here
for fold in folds: fit(model, train_idx); score(model, valid_idx)
THRESHOLD = 0.31  # chosen on out-of-fold scores, constraint recall >= 0.80
