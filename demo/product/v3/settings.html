Connect your tools

Let your coding agent run DCLab for you. It can read, upload and run experiments; it cannot release a model.

// .mcp.json
{ "mcpServers": { "dclab": {
    "command": "python", "args": ["-m", "dclab_mcp"],
    "env": { "DCLAB_TOKEN": "${DCLAB_TOKEN}" } } } }
from dclab_client import DCLab

lab = DCLab()                                   # reads DCLAB_TOKEN
project = lab.projects.create(name="Northwind churn")
data    = lab.datasets.upload(project.id, "customers.csv")
run     = lab.experiments.run(project.id, dataset=data.id, target="churned_next_30d")
print(lab.experiments.get(run.id).summary)
$ dclab data upload customers.csv --project northwind-churn
$ dclab experiments run --project northwind-churn --wait
$ dclab experiments compare E1 E7
$ dclab experiments code E7 > reproduce.py
