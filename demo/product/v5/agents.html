from dclab_client import DCLab
lab = DCLab(token=os.environ["DCLAB_TOKEN"])
p  = lab.projects.create(name="Northwind churn")
ds = lab.datasets.upload(p.id, "customers.csv")        # ingests + publishes
sp = lab.projects.create_problem_spec(p.id, target="churned_next_30d",
        constraints=[{"metric": "recall", "op": ">=", "value": 0.8}])
e1 = lab.experiments.run(p.id, dataset=ds.id, spec=sp.id)   # 202 + job
e2 = lab.experiments.branch(e1.id, changes=[{"kind": "class_weighting", "mode": "balanced"}])
print(lab.experiments.compare([e1.id, e2.id]))
print(lab.experiments.code(e2.id).script)
